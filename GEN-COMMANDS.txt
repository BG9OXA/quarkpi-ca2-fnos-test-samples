## [✅ 能播对照] 8bit H.264（-maxrate 控体积 ✓）

-ss 30 -t 20 -i ../4k-test/bbb_sunflower_2160p_30fps_normal.mp4 -vf scale=1920:1080 -c:v libx264 -preset veryfast -crf 20 -maxrate 2M -bufsize 4M -profile:v high -pix_fmt yuv420p -c:a aac -ac 2 -b:a 128k -movflags +faststart h264_8bit_1080p.mp4

-ss 30 -t 20 -i ../4k-test/bbb_sunflower_2160p_30fps_normal.mp4 -c:v libx264 -preset veryfast -crf 20 -maxrate 4M -bufsize 8M -profile:v high -pix_fmt yuv420p -c:a copy -movflags +faststart h264_8bit_4k.mp4
## [❌ 打不开] 10bit H.264

-ss 30 -t 20 -i ../4k-test/bbb_sunflower_2160p_30fps_normal.mp4 -vf scale=1920:1080,format=yuv420p10le -c:v libx264 -preset veryfast -crf 20 -maxrate 2M -bufsize 4M -profile:v high10 -pix_fmt yuv420p10le -c:a aac -ac 2 -b:a 128k -movflags +faststart h264_10bit_1080p.mp4

-ss 30 -t 20 -i ../4k-test/bbb_sunflower_2160p_30fps_normal.mp4 -vf scale=1920:1080,format=yuv420p10le -c:v libx265 -preset veryfast -crf 22 -maxrate 2M -bufsize 4M -profile:v main10 -c:a aac -ac 2 -b:a 128k -movflags +faststart hevc_10bit_1080p.mp4
## [诊断变体] 色彩标记 / 去音轨（流复制自主样本 ✓）

-i hevc_10bit_4k.mp4 -c copy -bsf:v hevc_metadata=transfer_characteristics=16:colour_primaries=9:matrix_coefficients=9 -color_trc smpte2084 -color_primaries bt2020 -colorspace bt2020nc -movflags +faststart v_hdr_pq.mp4

-i hevc_10bit_4k.mp4 -c copy -bsf:v hevc_metadata=transfer_characteristics=1:colour_primaries=1:matrix_coefficients=1 -color_trc bt709 -color_primaries bt709 -colorspace bt709 -movflags +faststart v_bt709.mp4

-i hevc_10bit_4k.mp4 -c copy -an -movflags +faststart v_noaudio.mp4
